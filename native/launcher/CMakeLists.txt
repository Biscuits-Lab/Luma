cmake_minimum_required(VERSION 3.20)
project(LumaLauncher LANGUAGES CXX)
option(LUMA_CONTRACT_ONLY "Build portable parser/quoting tests only; not Windows evidence" OFF)
if(NOT LUMA_CONTRACT_ONLY AND (NOT WIN32 OR NOT MSVC))
    message(FATAL_ERROR "Build the launcher with MSVC on Windows, or select LUMA_CONTRACT_ONLY for parser tests.")
endif()
add_library(launcher_contract STATIC launcher_contract.cpp)
target_compile_features(launcher_contract PUBLIC cxx_std_17)
include(CTest)
if(BUILD_TESTING)
    add_executable(launcher_contract_tests launcher_contract_tests.cpp)
    target_link_libraries(launcher_contract_tests PRIVATE launcher_contract)
    add_test(NAME launcher_contract COMMAND launcher_contract_tests)
endif()
if(NOT LUMA_CONTRACT_ONLY)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Luma launcher requires a Windows x64 build.")
    endif()
    enable_language(RC)
    get_filename_component(LUMA_LAUNCHER_ICON "${CMAKE_CURRENT_SOURCE_DIR}/../../luma/assets/icon.ico" ABSOLUTE)
    configure_file(launcher.rc.in "${CMAKE_CURRENT_BINARY_DIR}/launcher.rc" @ONLY)
    add_executable(LumaLauncher WIN32 launcher.cpp "${CMAKE_CURRENT_BINARY_DIR}/launcher.rc" launcher.manifest)
    target_link_libraries(LumaLauncher PRIVATE launcher_contract shell32 user32 bcrypt)
    set_target_properties(LumaLauncher PROPERTIES OUTPUT_NAME Luma)
    set(LUMA_NATIVE_TARGETS launcher_contract LumaLauncher)
    if(BUILD_TESTING)
        add_executable(launcher_fixture WIN32 launcher_fixture.cpp)
        target_link_libraries(launcher_fixture PRIVATE shell32)
        list(APPEND LUMA_NATIVE_TARGETS launcher_contract_tests launcher_fixture)
        find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
        add_test(NAME launcher_windows
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_launcher_windows.py"
                    --launcher "$<TARGET_FILE:LumaLauncher>" --fixture "$<TARGET_FILE:launcher_fixture>"
                    --report "${CMAKE_CURRENT_BINARY_DIR}/launcher-native-report.json")
        set_tests_properties(launcher_windows PROPERTIES TIMEOUT 180)
    endif()
    foreach(TARGET IN LISTS LUMA_NATIVE_TARGETS)
        target_compile_features(${TARGET} PRIVATE cxx_std_17)
        target_compile_definitions(${TARGET} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
        target_compile_options(${TARGET} PRIVATE /W4 /permissive- /EHsc /sdl /guard:cf /utf-8)
        target_link_options(${TARGET} PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
        set_property(TARGET ${TARGET} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endforeach()
    # Fixed launcher bytes across repeated builds of this protocol implementation.
    target_link_options(LumaLauncher PRIVATE /Brepro)
endif()
